Cấu Hình Cloudflare Worker (0đ)

Mặc định web gọi API công khai trực tiếp. Nếu muốn tốc độ nhanh & ổn định riêng tư 100%, dán URL Cloudflare Worker của bạn vào đây:

Mã nguồn worker.js (dán lên Cloudflare là chạy): 
export default {
  async fetch(req) {
    const u = new URL(req.url);
    const target = u.searchParams.get("url");
    const cors = {
      "Access-Control-Allow-Origin": "*",
      "Content-Type": "application/json"
    };
    if (!target) return new Response(JSON.stringify({error:"Missing url"}), {headers:cors});
    const api = "https://www.tikwm.com/api/?url=" + encodeURIComponent(target);
    const r = await fetch(api, { headers: { "User-Agent": "Mozilla/5.0" } });
    return new Response(await r.text(), { headers: cors });
  }
};
